        function greet() {
            const name = document.getElementById("nameInput").value.trim();
            const messageDiv = document.getElementById("message");

            if (name === "") {
                messageDiv.textContent = "Lütfen bir isim gir 🙂";
            } else {
                messageDiv.textContent = `Merhaba ${name}! Siteye hoş geldin 🎉`;
            }
        }
